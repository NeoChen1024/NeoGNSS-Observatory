find_package(OpenSSL REQUIRED COMPONENTS Crypto)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../contrib/json/include/nlohmann/json.hpp")
    message(FATAL_ERROR "Initialize contrib/json: git submodule update --init contrib/json")
endif()
add_library(neognss_obs src/ubx_archive.cpp src/reconstruction.cpp src/processing.cpp src/clock.cpp src/grid.cpp src/dataset.cpp)
add_library(neognss_obs::neognss_obs ALIAS neognss_obs)
target_compile_features(neognss_obs PUBLIC cxx_std_20)
set_target_properties(neognss_obs PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(neognss_obs PUBLIC include ../contrib/json/include)
target_link_libraries(neognss_obs PUBLIC cppgnss::cppgnss PRIVATE OpenSSL::Crypto)
option(NEOGNSS_BUILD_BINDINGS "Build Observatory Python batch processing bindings" ON)
if(NEOGNSS_BUILD_BINDINGS)
    if(NOT DEFINED Python_EXECUTABLE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../.venv/bin/python")
        set(Python_EXECUTABLE "${CMAKE_CURRENT_SOURCE_DIR}/../.venv/bin/python")
    endif()
    find_package(Python 3.11 REQUIRED COMPONENTS Interpreter Development.Module)
    execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
        OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(_native src/bindings.cpp)
    target_include_directories(_native PRIVATE ../contrib/json/include)
    target_link_libraries(_native PRIVATE neognss_obs::neognss_obs)
    if(DEFINED NEOGNSS_PYTHON_OUTPUT)
        set_target_properties(_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${NEOGNSS_PYTHON_OUTPUT}")
    endif()
endif()
